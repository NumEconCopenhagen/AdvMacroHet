\section[Pareto tails]{Pareto tails and random growth}

\begin{frame}{Pareto distribution}
\begin{itemize}
\item Wealth $a$ is Pareto with minimum $a_{0}$ and tail index $\zeta$ if
\[
P\left(a>x\right)=\left(\frac{x}{a_{0}}\right)^{-\zeta},\quad x\geq a_{0}
\]
\item Pareto tail: $P(a>x)\approx Cx^{-\zeta}$ for large $x$. Lower $\zeta$ = fatter tail

\end{itemize}
\end{frame}

\begin{frame}{Properties of the Pareto distribution}
\begin{itemize}
\item \textbf{P1. Moments:} $E\left[a^{\theta}\right]=\frac{\zeta a_{0}^{\theta}}{\zeta-\theta}$ if $\theta<\zeta$, $\infty$ otherwise (mean wealth is finite only if $\zeta>1$)
\item \textbf{P2. Van der Wijk's law:} above $x$, $a$ is Pareto with minimum $x$, so
\[
E\left[a\mid a>x\right]=\frac{\zeta}{\zeta-1}x
\]
\item \textbf{P3. Top shares:} the top fraction $q$ holds
\[
S(q)=q^{1-1/\zeta}
\]
\begin{itemize}
\item $\zeta=1.5$: top 1\% holds $0.01^{1/3}\approx22\%$
\end{itemize}
\item \textbf{Power rule:} if $X=a^{b}$ with $b>0$, then $\zeta_{X}=\zeta_{a}/b$
\begin{itemize}
\item proof: $P(X>x)=P(a>x^{1/b})\propto x^{-\zeta_{a}/b}$
\item concave functions of wealth ($b<1$) have thinner tails, convex ones ($b>1$) fatter tails
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Wealth as a random growth process}
\begin{itemize}
\item Budget constraint: $a_{it}=(1+r_{it})a_{it-1}+wz_{it}-c_{it}$
\item Let households save a share $s_{it}$ of cash-on-hand:
\[
a_{it}=G_{it}a_{it-1}+y_{it},\qquad G_{it}=s_{it}(1+r_{it}),\qquad y_{it}=s_{it}wz_{it}
\]
\item $G_{it}$: growth factor, scales with wealth
\item $y_{it}$: saved labor income, does not scale with wealth
\item Assume $(G_{it},y_{it})\geq0$ i.i.d. over time and households, independent of $a_{it-1}$
\item Two routes to a Pareto tail of wealth:
\begin{enumerate}
\item Pareto labor income: $y_{it}$ has a Pareto tail
\item Random growth: $G_{it}$ random
\end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Route 1: Pareto labor income}
\begin{itemize}
\item $G\in(0,1)$ constant, $y_{it}$ i.i.d. with a Pareto tail, $P(y>x)\approx Cx^{-\zeta_{y}}$
\item Wealth is a weighted sum of past incomes:
\[
a_{it}=\sum_{k\geq0}G^{k}y_{i,t-k}
\]
\item Large when one term is large:
\[
P(a_{it}>x)\approx\sum_{k}C\left(\frac{x}{G^{k}}\right)^{-\zeta_{y}}\propto x^{-\zeta_{y}}
\]
\item Wealth inherits the labor income tail: $\zeta_{a}=\zeta_{y}$
\item Data: $\zeta_{a}=1.38<\zeta_{y}=2.25$ $\Rightarrow$ labor income alone cannot explain the wealth tail, we need random growth (Stachurski and Toda, 2019)
\end{itemize}
\end{frame}

\begin{frame}[label=route2]{Route 2: random growth}
\begin{itemize}
\item $G_{it}$ i.i.d., random through returns or saving rates, with
\begin{itemize}
\item $P(G>1)>0$: some households grow by luck
\item $E[\ln G]<0$: something stops them (they shrink on average, or die)
\item $y_{it}$ thin-tailed (e.g. lognormal), positive with positive probability
\end{itemize}
\item One can show (Kesten, 1973) that wealth has a Pareto tail, $P(a>x)\approx Cx^{-\zeta}$, where $\zeta>0$ solves
\[
E\left[G^{\zeta}\right]=1
\]
\item Lognormal $G$, $\ln G\sim\mathcal{N}(\mu_{G},\sigma_{G}^{2})$ with $\mu_{G}<0$:
\[
\zeta=-\frac{2\mu_{G}}{\sigma_{G}^{2}}
\]
\begin{itemize}
\item $\zeta$ = pull back down ($-\mu_{G}$) relative to luck ($\sigma_{G}^{2}/2$): more luck or less shrinking gives a fatter tail
\end{itemize}
\end{itemize}
\hfill\hyperlink{route2-proof}{\beamerbutton{Proof}}
\end{frame}

\begin{frame}{Constant $\bar{G}$ + reset model}
\begin{itemize}
\item Survivors' wealth grows by a constant $\bar{G}>1$; each period households die with i.i.d. probability $\xi$ and are replaced by newborns with wealth $a_{0}$
\item Exact solution by age: wealth at age $s$ is $a_{0}\bar{G}^{s}$, and the share aged $s$ or more is $(1-\xi)^{s}$, so
\[
P(a>x)=P\left(\text{age}\geq\frac{\ln(x/a_{0})}{\ln\bar{G}}\right)=(1-\xi)^{\ln(x/a_{0})/\ln\bar{G}}=\left(\frac{x}{a_{0}}\right)^{-\zeta}
\]
\[
\zeta=-\frac{\ln(1-\xi)}{\ln\bar{G}}\approx\frac{\xi}{\ln\bar{G}}\qquad\Leftrightarrow\qquad(1-\xi)\bar{G}^{\zeta}=1
\]
\item Fatter tail when the rich grow faster or live longer. $\xi=0.02$, $\bar{G}=1.013$: $\zeta\approx1.56$
\end{itemize}
{\footnotesize Source: Jones (2015), \emph{Pareto and Piketty}, JEP}
\end{frame}

\begin{frame}{Growth, returns and taxes}
\begin{itemize}
\item Add economic growth: wages and average wealth grow at rate $g$, $w_{t}=(1+g)^{t}w_{0}$. Also add risky returns (mean $\bar{r}$, std. $\sigma_{r}$), consumption share of the rich $\bar{c}$, and taxes on capital income $\tau_{k}$, wealth $\tau_{w}$, labor income $\tau_{\ell}$:
\[
a_{it}=\left(1+(1-\tau_{k})r_{it}-\tau_{w}\right)a_{it-1}+(1-\tau_{\ell})w_{t}z_{it}-c_{it}
\]
\item Relative to the average, $\hat{a}_{it}=a_{it}/(1+g)^{t}$:
\[
\hat{a}_{it}=G_{it}\hat{a}_{it-1}+y_{it},\qquad G_{it}=\frac{(1-\bar{c})\left(1+(1-\tau_{k})r_{it}-\tau_{w}\right)}{1+g}
\]
\item Lognormal returns and $\ln(1+x)\approx x$ in $\zeta=-2\mu_{G}/\sigma_{G}^{2}$:
\[
\zeta\approx1-\frac{(1-\tau_{k})\bar{r}-\tau_{w}-g-\bar{c}}{(1-\tau_{k})^{2}\sigma_{r}^{2}/2}
\]
\end{itemize}
{\footnotesize Discrete-time version of Moll (2014), \emph{Why Piketty says r$-$g matters for inequality}}
\end{frame}

\begin{frame}{Piketty's $r-g$ and taxes}
\[
\zeta\approx1-\frac{(1-\tau_{k})\bar{r}-\tau_{w}-g-\bar{c}}{(1-\tau_{k})^{2}\sigma_{r}^{2}/2}
\]
\begin{itemize}
\item Top inequality $1/\zeta$ rises with $\bar{r}-g$: Piketty's (2014) $r>g$
\item But what matters is $\bar{r}-g-\bar{c}$: the return net of growth and of what the rich consume
\item Return risk is essential: without it, no Pareto tail. $r>g$ is neither necessary nor sufficient
\item $\zeta>1$ needs $(1-\tau_{k})\bar{r}-\tau_{w}-\bar{c}<g$: on average the rich grow slower than the economy, only the lucky ones climb
\item Taxes:
\begin{itemize}
\item wealth tax: lowers mean growth one for one
\item capital income tax: lowers mean growth and risk
\item labor income tax: only enters $y$, does not change $\zeta$
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Taking stock}

\begin{itemize}
    \item Labor income inequality alone cannot generate the relative Pareto tails of wealth / income
    \item Random growth mechanisms can generate the Pareto tails observed in the data
    \item The spread $r-g$ increases wealth inequality
\end{itemize}    

But we miss two key features:
\begin{itemize}
    \item Microfoundations: what determines $G$, etc?
    \item General equilibrium: $r$, $g$ are endogenous to behaviors and the economy
\end{itemize}
\end{frame}
\subsection{Failure of the buffer-stock model}

\begin{frame}{Aiyagari model}
\[
\max_{\left\{ c_{t}\right\} }E_{0}\sum_{t=0}^{\infty}\beta^{t}\frac{c_{t}^{1-\sigma}}{1-\sigma}\quad\text{s.t.}\quad a_{t}=(1+r)a_{t-1}+wz_{t}-c_{t},\quad a_{t}\geq0
\]
with $\ln z_{t}$ an AR(1) estimated on panel data (Floden and Lind\'e, 2001)
\begin{center}
\begin{tabular}{lcccc}
\toprule
 & Wealth Gini & Top 1\% & Top 5\% & Top 20\% \\
\midrule
US data (SCF, 1989) & 0.78 & 29 & 53 & 80 \\
Aiyagari baseline & 0.38 & 3.2 & 12.2 & 41.0 \\
\bottomrule
\end{tabular}
\end{center}
\pause
$\to$ misses the top by a factor of ten
\end{frame}

\begin{frame}{Policies in the buffer-stock model}
\begin{figure}
\centering
\includegraphics[width=0.7\linewidth]{figs/policy_functions_wide.pdf}
\end{figure}
\begin{itemize}
\item Above some wealth, every household dissaves, even the most productive
\item The top type stops at $a\approx640$, about 40 times average wealth: wealth is bounded, so no Pareto tail
\end{itemize}
{\footnotesize Lecture 4 HANC calibration (quarterly): $\sigma=1$, $\beta(1+r)=0.993$, $\rho_{z}=0.9$, $\sigma_{\psi}=0.5$}
\end{frame}

\begin{frame}{Why the buffer-stock model fails}
\begin{itemize}
\item Precautionary saving: once the buffer-stock target is reached, households dissave (Carroll, 1997)
\item At the top, $c=\bar{c}m$ and the Euler equation $c_{t}^{-\sigma}=\beta(1+r)c_{t+1}^{-\sigma}$ give
\[
G_{\text{top}}=(1+r)(1-\bar{c})=\left(\beta(1+r)\right)^{1/\sigma}<1
\]
since $\beta(1+r)<1$ in equilibrium
\item No route 1: an AR(1) gives labor income a thin tail
\item No route 2: $G<1$ and deterministic at the top
\item The rich dissave in the model, but save more in the data (Dynan, Skinner and Zeldes, 2004; De Nardi, French and Jones, 2010)
\end{itemize}
\end{frame}

\section[Explaining wealth inequality]{Explaining wealth inequality}

\begin{frame}{Four main mechanisms}
Each gives some households $G>1$:
\begin{enumerate}
\item \textbf{Bequests (or taste for wealth)}
\begin{itemize}
\item Kotlikoff and Summers (1981), Gale and Scholz (1994), De Nardi (2004)
\end{itemize}
\item \textbf{Heterogeneous preferences}
\begin{itemize}
\item Krusell and Smith (1998), Druedahl and Jørgensen (2015)
\end{itemize}
\item \textbf{Entrepreneurship}
\begin{itemize}
\item Quadrini (2000), Cagetti and De Nardi (2006), Guvenen et al. (2023)
\end{itemize}
\item \textbf{Idiosyncratic rates of returns}
\begin{itemize}
\item Benhabib, Bisin and Zhu (2011), Hubmer, Krusell and Smith (2021)
\end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Explaining wealth inequality}
\[
\begin{gathered}\max_{\left\{ c_{t}\right\} _{t=0}^{T}}E\sum_{t=0}^{T}\beta^{t}\left(\left(1-d_{t}\right)\frac{c_{t}^{1-\sigma}}{1-\sigma}+\textcolor{blue}{d_{t}\phi\left(a_{t-1}\right)}\right)\\
c_{t}+a_{t}=y_{t}+(1+r)a_{t-1}+\textcolor{blue}{b_{t}},\quad a_{t}\geq\underline{a}
\end{gathered}
\]
\begin{enumerate}
\item \textbf{Bequests (or taste for wealth)}: $d_{t}=1$ at death, warm glow $\phi(a)$, or $\kappa\frac{a^{1-\nu}}{1-\nu}$ with $\nu<\sigma$
\begin{itemize}
\item wealth is a luxury good: $\bar{c}\to0$ at the top, so $G_{\text{top}}\to1+r$
\item inheritances $b_{t}$ also weaken the reset at death
\end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Explaining wealth inequality}
\[
\begin{gathered}\max_{\left\{ c_{t}\right\} _{t=0}^{T}}E\sum_{t=0}^{T}\textcolor{blue}{\beta_{i}}^{t}\left(1-d_{t}\right)\frac{c_{t}^{1-\textcolor{blue}{\sigma_{i}}}}{1-\textcolor{blue}{\sigma_{i}}}\\
c_{t}+a_{t}=y_{t}+(1+r)a_{t-1},\quad a_{t}\geq\underline{a}
\end{gathered}
\]
\begin{enumerate}
\setcounter{enumi}{1}
\item \textbf{Heterogeneous preferences}: patient households have a lower $\bar{c}$
\begin{itemize}
\item random $\beta$ makes saving rates random (route 2)
\item often chosen residually to match wealth (lecture 4, HKS 2021)
\end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Explaining wealth inequality}

\begin{gathered}\max_{\left\{ c_{t}\right\} _{t=0}^{T}}E\sum_{t=0}^{T}\beta^{t}\left(1-d_{t}\right)\frac{c_{t}^{1-\sigma}}{1-\sigma}\\
c_{t}+a_{t}=\textcolor{blue}{\left[l_{e}f\left(\theta_{t},k_{t-1}\right)+\left(1-l_{e}\right)y_{t}\right]}+(1+r)\left(a_{t-1}-k_{t-1}\right),\quad a_{t}\geq\underline{a}

\begin{enumerate}
\setcounter{enumi}{2}
\item \textbf{Entrepreneurship}: high, risky, persistent returns on own wealth, $k\leq\lambda a$
\begin{itemize}
\item reduced form: "superstar" earnings state (Castañeda et al., 2003), which works through route 1
\end{itemize}
\end{enumerate}

\begin{enumerate}
\setcounter{enumi}{3}
\item \textbf{Idiosyncratic rates of returns}: random growth (route 2) directly
\end{enumerate}

\begin{frame}{Return heterogeneity: type vs scale dependence}
\begin{itemize}
\item \textbf{Type dependence:} $r(z_{it})$, the return depends on a persistent type
\begin{itemize}
\item capital income is linear in wealth within a type
\item $\zeta_{ra}=\zeta_{a}$ far in the tail
\end{itemize}
\item \textbf{Scale dependence:} $r\,S(a)$ with $S(a)=1+\psi a^{\eta}$
\begin{itemize}
\item the rich grow faster because they are rich
\item capital income $\propto a^{1+\eta}$, so $\zeta_{ra}=\zeta_{a}/(1+\eta)$
\end{itemize}
\item Returns rise with wealth in the data, but richer households may also be higher types
\begin{itemize}
\item panel data on returns help separate the two (Fagereng, Guiso, Malacrino and Pistaferri, 2020)
\end{itemize}
\end{itemize}
\end{frame}

\subsection{Gaillard, Hellwig, Wangner and Werquin (2024)}

\begin{frame}{Gaillard et al. (2024)}
\begin{itemize}
\item \textbf{Question:} many models generate high wealth inequality. Which ones match the ranking
\[
\zeta_{ra}<\zeta_{a}<\zeta_{y}<\zeta_{c}\qquad(1.20,\;1.38,\;2.25,\;3.06)
\]
\item \textbf{Answer:} two ingredients
\begin{enumerate}
\item non-homothetic preferences: decouple consumption from wealth
\item scale-dependent returns: decouple capital income from wealth
\end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Model}
\[

\end{aligned}
\]
\begin{itemize}
\item \textbf{Taste for wealth}, $\nu<\gamma$: wealth is a luxury good
\item \textbf{Return type} $z\in\{1,z_{H}\}$, worker or entrepreneur: enter with probability 0.02, exit with 0.2
\item \textbf{Scale dependence} $S(a)=1+\psi a^{\eta}$: the return per unit of wealth rises with wealth
\item Labor productivity: AR(1), with a Pareto tail ($\zeta_{h}=2.25$) at the top; HSV labor tax
\item Death $\xi=0.02$, bequests fully taxed: death resets wealth
\item Cobb-Douglas firms; capital is effective wealth, $K=\int zS(a)a\,dF$
\end{itemize}
\end{frame}

\begin{frame}{Intuition: two ingredients, two tails}
\begin{itemize}
\item Power rule: if $X\propto a^{b}$ at the top, then $\zeta_{X}=\zeta_{a}/b$
\begin{itemize}
\item concave in wealth: thinner tail; convex: fatter tail
\end{itemize}
\item \textbf{Consumption} needs to be concave in wealth to have $\zeta_{c}>\zeta_{a}$
\begin{itemize}
\item homothetic preferences: the rich consume a constant share of wealth, $c\propto a$, so $\zeta_{c}=\zeta_{a}$
\item taste for wealth: the rich save most of their income, $c$ is concave in $a$, so $\zeta_{c}>\zeta_{a}$
\end{itemize}
\item \textbf{Capital income} needs to be convex in wealth to have $\zeta_{ra}<\zeta_{a}$
\begin{itemize}
\item same return for all: capital income $\propto a$, so $\zeta_{ra}=\zeta_{a}$
\item scale dependence: capital income $\propto a^{1+\eta}$, so $\zeta_{ra}=\zeta_{a}/(1+\eta)$
\end{itemize}
\end{itemize}
\end{frame}

\begin{itemize}

\item Without return heterogeneity, a wealth tax and a capital income tax are equivalent:

a+(1-\tau_{k})ra=(1-\tau_{w})a+ra\quad\text{if}\quad\tau_{w}=r\tau_{k}

\begin{itemize}

\item wealth tax: same tax for the same wealth, shifts the burden to unproductive households
\end{itemize}
\end{itemize}

\item Entrepreneurial profits: $\pi(a,z)=\max_{k\leq\lambda a}\left\{ pzk-(r+\delta)k\right\} $

\subsection{Hubmer, Krusell and Smith (2021)}

\begin{frame}{Hubmer, Krusell and Smith (2021)}
\begin{itemize}

\item \textbf{Question:} can the observed changes in taxes, earnings and asset returns since 1967 explain the rise in US wealth inequality?

\begin{enumerate}

\item counterfactuals: one change at a time
\end{enumerate}
\end{itemize}

\begin{frame}{Household problem}
\[
\begin{aligned}v_{t}(\omega_{t},a_{t-1})= & \max_{a_{t}\geq\underline{a}}\frac{c_{t}^{1-\gamma}}{1-\gamma}+\textcolor{blue}{\beta_{t}}E_{t}\left[v_{t+1}(\omega_{t+1},a_{t})\right]\\

\end{aligned}
\]
{\small

\item $y_{t}$: labor income and expected capital income, taxed progressively by $\tau_{t}(\cdot)$
\item $\tilde{y}_{t}$: random capital income, $\eta_{t}\sim\mathcal{N}(0,1)$ i.i.d., taxed at a flat rate $\tilde{\tau}_{t}$
\item \textbf{Returns} $r_{t}^{X}(a)$, $\sigma^{X}(a)$ rise with wealth (Swedish portfolio data)

\item \textbf{capital income}: what households earn equals what firms pay

\big(F_{K}(K_{t},L)-\delta\big)K_{t}=\int\big(r_{t}+r_{t}^{X}(a)\big)a\,d\Gamma_{t}

\end{enumerate}

\subsection{Moll, Rachel and Restrepo (2022)}

\begin{frame}{Moll, Rachel and Restrepo (2022): model}
\textbf{Households} with skill $z$ (wage percentile, population share $\ell_{z}$), continuous time:
\[
\max\int_{0}^{\infty}e^{-\rho t}\frac{c_{t}^{1-\sigma}}{1-\sigma}dt\quad\text{s.t.}\quad\dot{a}_{t}=w_{z}+ra_{t}-c_{t}
\]
\vspace{-4mm}
\begin{itemize}
\item same return $r$ for everyone, no income risk
\item \textcolor{blue}{dissipation shock} at rate $\xi$: wealth reset to 0 (death without bequest, failed business, \ldots); $\rho$ includes $\xi$
\end{itemize}
\textbf{Firms}: tasks done by labor or capital; a share $\alpha_{z}$ of the tasks of skill $z$ is \textcolor{blue}{automated}
\[
Y=AK^{\alpha}\prod_{z}(\psi_{z}\ell_{z})^{(1-\alpha_{z})\eta_{z}},\qquad\alpha=\sum_{z}\alpha_{z}\eta_{z}
\]
\vspace{-4mm}
\begin{itemize}
\item factor prices: $w_{z}\ell_{z}=(1-\alpha_{z})\eta_{z}Y$ and $r+\delta=\alpha Y/K$
\item \textbf{automation}: $\alpha_{z}$ rises, tasks move from skill $z$ to capital
\end{itemize}
\textbf{Equilibrium}: $K$ = household wealth, summed over skills and over time since the last reset
\end{frame}

\begin{frame}{Key result: top inequality = net capital share}
\begin{itemize}
\item Saving is linear in total wealth $x=a+w/r$, which grows at $g=\frac{r-\rho}{\sigma}$: the constant $\bar{G}$ + reset model, so
\[
\frac{1}{\zeta}=\frac{g}{\xi}
\]
\item Steady state: wealth added by savers = wealth lost to resets, with $\bar{w}=\sum_{z}w_{z}\ell_{z}$ the wage bill
\[
g\Big(K+\frac{\bar{w}}{r}\Big)=\xi K\quad\Rightarrow\quad\frac{1}{\zeta}=\frac{g}{\xi}=\frac{rK}{rK+\bar{w}}=\alpha_{net}
\]
\item \textbf{Top inequality = net capital share}
\begin{itemize}
\item all capital is rebuilt from savings: the more capital the economy needs, the faster savers must grow, the fatter the tail
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Application: automation}
\begin{itemize}
\item Automation shifts tasks from labor to capital: more demand for capital
\begin{itemize}
\item $r$ and $\alpha_{net}$ rise: fortunes grow faster, fatter tail
\item part of the productivity gain goes to capital: wages can fall
\end{itemize}
\item Experiment: automation of routine jobs, 1980--2014, calibrated to the rise of the capital share from 34.5\% to 42.8\%
\item Results (steady states of the extended model, with risky capital and growth, so $1/\zeta$ is no longer exactly $\alpha_{net}$):
\begin{itemize}
\item top income tail $1/\zeta$ from 0.54 to 0.65, about 70\% of the observed rise
\item wages fall below the 80th percentile; incomes $+20$ to $+55\%$ within the top 1\%, mostly capital income
\end{itemize}
\item Caveat: like all random growth models, the transition is too slow (Gabaix et al.)
\end{itemize}
\end{frame}

\begin{frame}{Exercise: HANC model with return heterogeneity}
\[
\begin{aligned}v_{t}(e_{it},r_{it}^{X},a_{it-1})= & \max_{c_{it},a_{it}}\frac{c_{it}^{1-\sigma}}{1-\sigma}+\beta E_{t}\left[v_{t+1}(e_{it+1},r_{it+1}^{X},a_{it})\right]\\
\text{s.t.}\quad & a_{it}+c_{it}=(1+r_{t}+r_{it}^{X})a_{it-1}+w_{t}e_{it},\qquad a_{it}\geq0\\

\begin{frame}[label=route2-proof]{Appendix: why $E[G^{\zeta}]=1$}
\small
\textbf{Guess and verify:} guess $P(a>x)\approx Cx^{-\zeta}$ for large $x$
\begin{enumerate}
\item \textbf{Who is above $x$ tomorrow?} For large $x$, $y$ is negligible, so $a_{it}>x$ iff $a_{it-1}>x/G_{it}$
\item \textbf{Pareto scaling:} with the guess, the number above $x/G$ is $G^{\zeta}$ times the number above $x$ (only a power law scales this way)
\[
P\left(a>x/G\right)\approx C\left(\frac{x}{G}\right)^{-\zeta}=G^{\zeta}\,Cx^{-\zeta}
\]
\item \textbf{Average over $G$} (independent of $a_{it-1}$):
\[
P(a_{it}>x)\approx E\left[G^{\zeta}\right]\,P(a_{it-1}>x)
\]
$E[G^{\zeta}]$: factor by which the number of households above $x$ changes in one period
\item \textbf{Stationarity:} this number must not change, at any large $x$: $\boxed{E\left[G^{\zeta}\right]=1}$
\end{enumerate}
{Lognormal $G$, $\ln G\sim\mathcal{N}(\mu_{G},\sigma_{G}^{2})$: $E[G^{\zeta}]=e^{\zeta\mu_{G}+\zeta^{2}\sigma_{G}^{2}/2}=1$, so $\zeta=-2\mu_{G}/\sigma_{G}^{2}$}

\end{frame}
